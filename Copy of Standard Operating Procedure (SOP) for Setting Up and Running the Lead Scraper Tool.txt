import tkinter as tk
from tkinter import messagebox
import requests, json, csv, re, time
from datetime import datetime


def extract_emails_from_url(website_url):
    """Extracts email addresses from the given website URL using regex."""
    try:
        resp = requests.get(website_url, timeout=10)
        if resp.status_code != 200:
            return []
        html = resp.text
        # Regex pattern for emails
        emails = set(re.findall(r"[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}", html))
        return list(emails)
    except Exception as e:
        print(f"Error fetching {website_url}: {e}")
        return []


def run_query():
    # Retrieve user inputs
    api_key = api_key_entry.get().strip()
    business_type = business_type_entry.get().strip()
    zip_code = zip_entry.get().strip()
    try:
        radius = int(radius_entry.get().strip())
    except ValueError:
        messagebox.showerror("Invalid Input", "Search radius must be an integer.")
        return
    custom_filename = file_name_entry.get().strip()
    
    if not api_key or not business_type or not zip_code:
        messagebox.showerror("Missing Input", "Please fill in the API Key, Business Type, and Zip Code fields.")
        return


    # Geocode the zip code using the Google Geocoding API
    geocode_url = f'https://maps.googleapis.com/maps/api/geocode/json?address={zip_code}&key={api_key}'
    geocode_response = requests.get(geocode_url)
    geocode_data = geocode_response.json()
    
    if geocode_data['status'] != 'OK':
        messagebox.showerror("Geocoding Error", "Geocoding API failed for the given zip code.")
        return
    location_data = geocode_data['results'][0]['geometry']['location']
    latitude = location_data['lat']
    longitude = location_data['lng']
    location = f"{latitude},{longitude}"
    
    # Build the initial URL for the Places API Nearby Search
    places_url = f'https://maps.googleapis.com/maps/api/place/nearbysearch/json?location={location}&radius={radius}&keyword={business_type}&key={api_key}'
    
    all_results = []
    response = requests.get(places_url)
    data = response.json()
    all_results.extend(data.get('results', []))
    
    # Handle pagination (up to a maximum of 60 results across 3 pages)
    next_page_token = data.get('next_page_token')
    while next_page_token:
        time.sleep(2)  # Pause to allow token activation
        paged_url = f"https://maps.googleapis.com/maps/api/place/nearbysearch/json?pagetoken={next_page_token}&key={api_key}"
        paged_response = requests.get(paged_url)
        paged_data = paged_response.json()
        all_results.extend(paged_data.get('results', []))
        next_page_token = paged_data.get('next_page_token')
    
    # Create a unique filename using the current date and time.
    timestamp = datetime.now().strftime('%Y%m%d-%H%M%S')
    if custom_filename == "":
        csv_filename = f"{business_type}_{zip_code}_{timestamp}.csv"
    else:
        csv_filename = f"{custom_filename}_{timestamp}.csv"
    
    # Write the collected data to a CSV file.
    with open(csv_filename, 'w', newline='', encoding='utf-8') as csvfile:
        fieldnames = ['Name', 'Phone', 'Address', 'Website', 'Emails', 'Rating', 'Google Reviews', 'Price Level', 'Types']
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
        writer.writeheader()
        
        for result in all_results:
            name = result.get('name', 'N/A')
            address = result.get('vicinity', 'N/A')
            place_id = result.get('place_id', '')
            
            # Request detailed info for each business
            details_url = (
                f'https://maps.googleapis.com/maps/api/place/details/json?place_id={place_id}'
                f'&fields=name,international_phone_number,formatted_address,website,rating,'
                f'user_ratings_total,price_level,types&key={api_key}'
            )
            details_response = requests.get(details_url)
            details_data = details_response.json()
            details_result = details_data.get('result', {})
            
            phone = details_result.get('international_phone_number', 'N/A')
            website = details_result.get('website', 'N/A')
            rating = details_result.get('rating', 'N/A')
            google_reviews = details_result.get('user_ratings_total', 'N/A')
            price_level = details_result.get('price_level', 'N/A')
            types = ', '.join(details_result.get('types', []))
            
            # Extract emails from the website (if available)
            emails_found = []
            if website != 'N/A':
                emails_found = extract_emails_from_url(website)
            emails_str = ', '.join(emails_found) if emails_found else 'N/A'
            
            writer.writerow({
                'Name': name,
                'Phone': phone,
                'Address': address,
                'Website': website,
                'Emails': emails_str,
                'Rating': rating,
                'Google Reviews': google_reviews,
                'Price Level': price_level,
                'Types': types
            })
    
    messagebox.showinfo("Success", f"Data successfully saved to {csv_filename}")
